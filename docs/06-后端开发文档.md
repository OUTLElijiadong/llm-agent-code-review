## 3. 分层职责

| 层 | 文件 | 职责 | 禁止 |
| --- | --- | --- | --- |
| API | `app/api/v1/*.py` | 路由声明、参数解析、调用 Service、返回 | 业务逻辑、SQL |
| Service | `app/services/*.py` | 业务编排、权限校验、事务边界 | HTTP / 框架细节 |
| Model | `app/models/*.py` | ORM 实体定义、字段、关系 | 业务方法(可加 helper) |
| Schema | `app/schemas/*.py` | 请求/响应模型、字段校验 | 数据库访问 |
| AI | `app/ai/*.py` | LLM 调用、Prompt、多 Agent 编排、解析 | 业务表写入(`DeepSeekAgent` 仅写 `ai_call_log`) |
| Exporter | `app/exporters/*.py` | 文档渲染 | 业务规则 |

## 5. 安全与依赖

### 5.1 `app/core/security.py`

```python
from datetime import datetime, timedelta, timezone
import jwt
from passlib.context import CryptContext
from app.core.config import settings


pwd_ctx = CryptContext(schemes=["bcrypt"], deprecated="auto", bcrypt__rounds=12)


def hash_password(plain: str) -> str:
    return pwd_ctx.hash(plain)


def verify_password(plain: str, hashed: str) -> bool:
    return pwd_ctx.verify(plain, hashed)


def create_access_token(user_id: int, role: str) -> str:
    now = datetime.now(timezone.utc)
    payload = {
        "sub": str(user_id),
        "role": role,
        "iat": int(now.timestamp()),
        "exp": int((now + timedelta(seconds=settings.jwt_expire_seconds)).timestamp()),
    }
    return jwt.encode(payload, settings.jwt_secret, algorithm=settings.jwt_algorithm)


def decode_token(token: str) -> dict:
    return jwt.decode(token, settings.jwt_secret, algorithms=[settings.jwt_algorithm])
```

### 5.2 `app/core/dependencies.py`

```python
from fastapi import Depends, Header
from sqlalchemy.orm import Session

from app.core.database import get_db
from app.core.security import decode_token
from app.core.exceptions import AuthError, ForbiddenError
from app.models.user import User


def get_current_user(
    authorization: str = Header(...),
    db: Session = Depends(get_db),
) -> User:
    if not authorization.startswith("Bearer "):
        raise AuthError("缺少 token", code=40100)
    token = authorization[7:]
    try:
        payload = decode_token(token)
    except Exception:
        raise AuthError("token 非法或已过期", code=40101)

    user = db.get(User, int(payload["sub"]))
    if not user or user.status != 1:
        raise ForbiddenError("账号不存在或已禁用", code=40301)
    return user


def require_admin(user: User = Depends(get_current_user)) -> User:
    if user.role != "admin":
        raise ForbiddenError("需要管理员权限", code=40300)
    return user
```

### 5.4 `app/core/error_handlers.py`

```python
from fastapi import FastAPI, Request
from fastapi.exceptions import RequestValidationError
from fastapi.responses import JSONResponse
from loguru import logger

from app.core.exceptions import AppError


def register_handlers(app: FastAPI):
    @app.exception_handler(AppError)
    async def app_error_handler(req: Request, exc: AppError):
        return JSONResponse(
            status_code=exc.http_status,
            content={
                "code": exc.code,
                "message": exc.message,
                "detail": exc.detail,
                "request_id": req.headers.get("X-Request-Id"),
            },
        )

    @app.exception_handler(RequestValidationError)
    async def validation_handler(req: Request, exc: RequestValidationError):
        return JSONResponse(
            status_code=400,
            content={
                "code": 40002,
                "message": "参数校验失败",
                "detail": exc.errors(),
                "request_id": req.headers.get("X-Request-Id"),
            },
        )

    @app.exception_handler(Exception)
    async def unhandled_handler(req: Request, exc: Exception):
        logger.exception(exc)
        return JSONResponse(
            status_code=500,
            content={
                "code": 50000,
                "message": "服务器内部错误",
                "detail": None,
                "request_id": req.headers.get("X-Request-Id"),
            },
        )
```

### 8.2 Auth 路由示例

```python
# app/api/v1/auth.py
from fastapi import APIRouter, Depends
from sqlalchemy.orm import Session

from app.core.database import get_db
from app.core.dependencies import get_current_user
from app.schemas.common import Resp
from app.schemas.auth import RegisterIn, LoginIn, LoginOut, UserOut
from app.services import auth_service
from app.models.user import User

router = APIRouter()


@router.post("/register", response_model=Resp[UserOut])
def register(payload: RegisterIn, db: Session = Depends(get_db)):
    user = auth_service.register(db, payload)
    return Resp(data=UserOut.model_validate(user))


@router.post("/login", response_model=Resp[LoginOut])
def login(payload: LoginIn, db: Session = Depends(get_db)):
    token, user = auth_service.login(db, payload.username, payload.password)
    return Resp(data=LoginOut(
        access_token=token,
        expires_in=7 * 24 * 3600,
        user=UserOut.model_validate(user),
    ))


@router.get("/me", response_model=Resp[UserOut])
def me(user: User = Depends(get_current_user)):
    return Resp(data=UserOut.model_validate(user))
```

from app.core.security import hash_password, verify_password, create_access_token
from app.models.user import User
from app.schemas.auth import RegisterIn


def register(db: Session, payload: RegisterIn) -> User:
    exists = db.query(User.id).filter(User.username == payload.username).first()
    if exists:
        raise ConflictError("用户名已存在", code=40901)
    user = User(
        username=payload.username,
        password=hash_password(payload.password),
        email=payload.email,
        nickname=payload.nickname or payload.username,
        role="user",
        status=1,
    )
    db.add(user)
    db.commit()
    db.refresh(user)
    return user


def login(db: Session, username: str, password: str):
    user = db.query(User).filter(User.username == username).first()
    if not user or not verify_password(password, user.password):
        raise AuthError("用户名或密码错误", code=40001)
    if user.status != 1:
        raise ForbiddenError("账号已禁用", code=40301)
    user.last_login = datetime.now(timezone.utc)
    db.commit()
    token = create_access_token(user.id, user.role)
    return token, user
```

from sqlalchemy.orm import Session
from fastapi import UploadFile
from typing import Optional

from app.core.config import settings
from app.core.exceptions import ValidationError, NotFoundError, ForbiddenError
from app.models.project import Project
from app.models.code_file import CodeFile
from app.models.code_version import CodeVersion
from app.utils.file_validator import validate_filename, validate_size
from app.utils.encoding_utils import to_utf8
from app.ai.language_detector import detect_language


def upload(db: Session, *, user, project_id: int, upload_file: UploadFile,
           file_path: Optional[str], language: Optional[str]):
    project = db.get(Project, project_id)
    if not project or project.status == "deleted":
        raise NotFoundError("项目不存在", code=40400)
    if project.user_id != user.id and user.role != "admin":
        raise ForbiddenError("无访问权限", code=40300)

    raw = upload_file.file.read()
    validate_size(len(raw), settings.max_upload_size)
    safe_name = validate_filename(upload_file.filename)

    text = to_utf8(raw)
    lang = language or detect_language(safe_name)

    file = CodeFile(
        project_id=project.id,
        file_name=safe_name,
        file_path=file_path or safe_name,
        language=lang,
        size_bytes=len(text.encode("utf-8")),
        line_count=text.count("\n") + 1,
        version_no=1,
        content=text,
        status="active",
    )
    db.add(file)
    db.flush()
    db.add(CodeVersion(
        file_id=file.id,
        version_no=1,
        content=text,
        change_desc="初始上传",
        operator_id=user.id,
    ))
    db.commit()
    db.refresh(file)
    return file.id, lang, 1


def update_content(db: Session, *, user, file_id: int, content: str, change_desc: Optional[str]):
    f = db.get(CodeFile, file_id)
    if not f or f.status == "deleted":
        raise NotFoundError("文件不存在", code=40400)
    # 权限校验
    project = db.get(Project, f.project_id)
    if project.user_id != user.id and user.role != "admin":
        raise ForbiddenError("无访问权限", code=40300)

    f.content = content
    f.size_bytes = len(content.encode("utf-8"))
    f.line_count = content.count("\n") + 1
    f.version_no += 1
    db.add(CodeVersion(
        file_id=f.id,
        version_no=f.version_no,
        content=content,
        change_desc=change_desc or "更新",
        operator_id=user.id,
    ))
    db.commit()
    return f.version_no
```

### 9.3 ReviewService(骨架)

```python
# app/services/review_service.py(骨架,具体 AI 调用在 ai/ 包)
import time
from datetime import datetime, timezone

from sqlalchemy.orm import Session
from loguru import logger

from app.core.exceptions import ValidationError, NotFoundError
from app.models.project import Project
from app.models.code_file import CodeFile
from app.models.review_task import ReviewTask
from app.models.review_issue import ReviewIssue
from app.schemas.review import ReviewStartIn

from app.services.rule_service import get_enabled_rules
from app.ai.prompt_builder import build_prompt
from app.ai.deepseek_agent import DeepSeekAgent
from app.ai.result_parser import parse
from app.ai.code_chunker import chunk_code
from app.ai.scoring import compute_score


def start(db: Session, *, user, payload: ReviewStartIn) -> ReviewTask:
    project = db.get(Project, payload.project_id)
    if not project:
        raise NotFoundError("项目不存在", code=40400)
    if not 1 <= len(payload.file_ids) <= 50:
        raise ValidationError("file_ids 需为 1-50 个", code=40001)

    files = (
        db.query(CodeFile)
        .filter(CodeFile.id.in_(payload.file_ids), CodeFile.status == "active")
        .all()
    )
    if len(files) != len(payload.file_ids):
        raise NotFoundError("部分文件不存在", code=40400)

    rules = get_enabled_rules(db, user.id)

    task = ReviewTask(
        user_id=user.id,
        project_id=project.id,
        task_name=payload.task_name or f"{project.project_name}-审查",
        review_type=payload.review_type,
        status="running",
        total_files=len(files),
        model_name=settings.deepseek_model,
        rules_snapshot=[r.rule_code for r in rules],
        start_time=datetime.now(timezone.utc),
    )
    db.add(task)
    db.commit()
    db.refresh(task)

    agent = DeepSeekAgent()
    t0 = time.time()
    try:
        all_issues: list[ReviewIssue] = []
        for f in files:
            file_issues = _review_one_file(db, agent, task, f, rules, user)
            all_issues.extend(file_issues)
            task.processed_files += 1
            db.commit()

        sev_count = {"严重": 0, "高": 0, "中": 0, "低": 0}
        for it in all_issues:
            sev_count[it.severity] = sev_count.get(it.severity, 0) + 1
        task.total_issues = len(all_issues)
        task.severe_issues = sev_count["严重"]
        task.high_issues = sev_count["高"]
        task.medium_issues = sev_count["中"]
        task.low_issues = sev_count["低"]
        task.score = compute_score(sev_count)
        task.status = "success"
        task.end_time = datetime.now(timezone.utc)
        task.duration_ms = int((time.time() - t0) * 1000)
        db.commit()
    except Exception as e:
        logger.exception(e)
        task.status = "failed"
        task.error_message = str(e)[:500]
        task.end_time = datetime.now(timezone.utc)
        task.duration_ms = int((time.time() - t0) * 1000)
        db.commit()
        raise

    db.refresh(task)
    return task


def _review_one_file(db, agent: DeepSeekAgent, task: ReviewTask, file: CodeFile, rules, user):
    chunks = chunk_code(file.content, file.language, threshold=6000)
    issues_acc: list[ReviewIssue] = []
    for idx, chunk in enumerate(chunks):
        system_prompt, user_prompt = build_prompt(
            language=file.language,
            file_name=file.file_name,
            code=chunk.text,
            rules=rules,
            line_offset=chunk.start_line,
        )
        try:
            raw, meta = agent.chat(
                system_prompt=system_prompt,
                user_prompt=user_prompt,
                task_id=task.id,
                user_id=user.id,
                file_id=file.id,
                chunk_index=idx,
                db=db,
            )
            result = parse(raw)
        except Exception as e:
            logger.warning(f"文件 {file.file_name} chunk {idx} 失败: {e}")
            continue

        for it in result.issues:
            issues_acc.append(ReviewIssue(
                task_id=task.id,
                file_id=file.id,
                file_name=file.file_name,
                line_number=(it.line_number or 0) + chunk.start_line,
                issue_type=it.issue_type,
                severity=it.severity,
                title=it.title,
                description=it.description,
                suggestion=it.suggestion,
                fixed_code=it.fixed_code,
                status="unfixed",
            ))

    if issues_acc:
        db.add_all(issues_acc)
        db.commit()
    return issues_acc
```

# app/utils/file_validator.py
import os
import re
from app.core.config import settings
from app.core.exceptions import ValidationError


BLOCK_DIRS = re.compile(
    r"(^|/)(\.git|\.svn|\.hg|node_modules|__pycache__|\.next|dist|build|target|\.idea|\.vscode)(/|$)"
)


def validate_filename(name: str, allowed_extensions: list[str]) -> str:
    base = os.path.basename(name).strip()
    if not base or len(base) > 255:
        raise ValidationError("文件名不合法", code=40001)
    if "\x00" in base:
        raise ValidationError("文件名包含非法字符(null)", code=40001)
    if BLOCK_DIRS.search(name.replace("\\", "/")):
        raise ValidationError(f"不允许上传系统目录中的文件: {name}", code=40001)
    ext = os.path.splitext(base)[1].lower()
    if ext and ext not in allowed_extensions and "*" not in allowed_extensions:
        raise ValidationError(f"不支持的文件类型 {ext}", code=41500)
    return base


def validate_size(size: int, limit: int):
    if size > limit:
        raise ValidationError(f"文件超过 {limit} 字节限制", code=41301)
```

# app/utils/encoding_utils.py
import chardet
from app.core.exceptions import ValidationError


def to_utf8(raw: bytes) -> str:
    if not raw:
        return ""
    try:
        return raw.decode("utf-8")
    except UnicodeDecodeError:
        guess = chardet.detect(raw)
        enc = guess.get("encoding") or "gbk"
        try:
            return raw.decode(enc, errors="replace")
        except Exception:
            raise ValidationError("无法识别文件编码,请保存为 UTF-8", code=40001)
```

## 12. 编码规范

- 模块名:`snake_case.py`
- 类名:`PascalCase`
- 函数/变量:`snake_case`
- 常量:`UPPER_SNAKE`
- 严禁使用 `print`,日志走 `loguru`
- 函数单一职责,签名带类型注解
- 数据库写操作必须事务包裹
- 异常向上抛 `AppError` 子类,被全局 handler 转 JSON
- 路由层不直接 `db.commit()`
- 严禁字符串拼 SQL

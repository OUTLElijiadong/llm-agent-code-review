## 4. HTTP 封装

### 4.1 `src/api/http.ts`

```ts
import axios, { AxiosError, AxiosResponse } from 'axios'
import { ElMessage } from 'element-plus'
import router from '@/router'
import { getToken, clearToken } from '@/utils/token'

export interface Resp<T = unknown> {
  code: number
  message: string
  data: T
  request_id?: string
  detail?: unknown
}

const http = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL,
  timeout: 60_000,
})

http.interceptors.request.use((config) => {
  const token = getToken()
  if (token) config.headers.Authorization = `Bearer ${token}`
  return config
})

http.interceptors.response.use(
  (resp: AxiosResponse<Resp>) => {
    const { data } = resp
    if (data?.code === 0) return resp
    ElMessage.error(data?.message || '请求失败')
    return Promise.reject(data)
  },
  (err: AxiosError<Resp>) => {
    const status = err.response?.status
    const data = err.response?.data
    if (status === 401) {
      clearToken()
      router.push('/login')
    }
    ElMessage.error(data?.message || err.message || '网络错误')
    return Promise.reject(data || err)
  },
)

export default http

export async function get<T>(url: string, params?: object): Promise<T> {
  const r = await http.get<Resp<T>>(url, { params })
  return r.data.data
}

export async function post<T>(url: string, body?: object, params?: object): Promise<T> {
  const r = await http.post<Resp<T>>(url, body, { params })
  return r.data.data
}

export async function put<T>(url: string, body?: object): Promise<T> {
  const r = await http.put<Resp<T>>(url, body)
  return r.data.data
}

export async function del<T>(url: string): Promise<T> {
  const r = await http.delete<Resp<T>>(url)
  return r.data.data
}

export function download(url: string, params?: object) {
  return http.get(url, { params, responseType: 'blob' })
}
```

### 5.2 守卫

```ts
// src/router/guards.ts
import { Router } from 'vue-router'
import { useUserStore } from '@/stores/user'

export function setupGuards(router: Router) {
  router.beforeEach(async (to) => {
    const user = useUserStore()
    if (to.meta.public) return true
    if (!user.token) return { path: '/login', query: { redirect: to.fullPath } }
    if (!user.profile) {
      try { await user.fetchProfile() } catch { return { path: '/login' } }
    }
    if (to.meta.role && to.meta.role !== user.profile?.role) return { path: '/403' }
    return true
  })
}
```

## 6. Pinia Stores

### 6.1 用户 Store

```ts
// src/stores/user.ts
import { defineStore } from 'pinia'
import { authApi } from '@/api/auth'
import type { UserOut } from '@/types/auth'
import { getToken, setToken, clearToken } from '@/utils/token'

export const useUserStore = defineStore('user', {
  state: () => ({
    token: getToken() as string | null,
    profile: null as UserOut | null,
  }),
  actions: {
    async login(username: string, password: string) {
      const r = await authApi.login({ username, password })
      this.token = r.access_token
      this.profile = r.user
      setToken(r.access_token)
    },
    async fetchProfile() {
      this.profile = await authApi.me()
    },
    logout() {
      this.token = null
      this.profile = null
      clearToken()
    },
    isAdmin(): boolean { return this.profile?.role === 'admin' },
  },
})
```

### 7.2 Monaco 编辑器组件

```vue
<!-- src/components/editor/MonacoEditor.vue -->
<template>
  <div ref="container" class="monaco-container" :style="{ height }"></div>
</template>

<script setup lang="ts">
import { onMounted, onBeforeUnmount, ref, watch, shallowRef } from 'vue'
import * as monaco from 'monaco-editor'

interface Props {
  modelValue: string
  language: string
  readonly?: boolean
  height?: string
  highlightLines?: number[]
}

const props = withDefaults(defineProps<Props>(), {
  readonly: false,
  height: '60vh',
  highlightLines: () => [],
})
const emit = defineEmits<{ (e: 'update:modelValue', value: string): void }>()

const container = ref<HTMLDivElement>()
const editor = shallowRef<monaco.editor.IStandaloneCodeEditor>()
const decorations = ref<string[]>([])

onMounted(() => {
  editor.value = monaco.editor.create(container.value!, {
    value: props.modelValue,
    language: props.language,
    automaticLayout: true,
    readOnly: props.readonly,
    minimap: { enabled: true },
    fontSize: 14,
    tabSize: 4,
    lineNumbers: 'on',
    scrollBeyondLastLine: false,
    theme: 'vs',
  })
  editor.value.onDidChangeModelContent(() => {
    emit('update:modelValue', editor.value!.getValue())
  })
  applyHighlight()
})

watch(() => props.modelValue, (val) => {
  if (editor.value && val !== editor.value.getValue()) {
    editor.value.setValue(val)
  }
})
watch(() => props.language, (lang) => {
  if (editor.value) monaco.editor.setModelLanguage(editor.value.getModel()!, lang)
})
watch(() => props.highlightLines, applyHighlight, { deep: true })

function applyHighlight() {
  if (!editor.value) return
  const next = (props.highlightLines || []).map((ln) => ({
    range: new monaco.Range(ln, 1, ln, 1),
    options: { isWholeLine: true, className: 'monaco-issue-line', glyphMarginClassName: 'monaco-issue-glyph' },
  }))
  decorations.value = editor.value.deltaDecorations(decorations.value, next)
}

onBeforeUnmount(() => editor.value?.dispose())
</script>

<style>
.monaco-container { border: 1px solid #dcdfe6; border-radius: 4px; }
.monaco-issue-line { background: rgba(245, 108, 108, 0.12); }
.monaco-issue-glyph { background: #f56c6c; }
</style>
```

### 7.3 代码编辑页

```vue
<!-- src/views/code/CodeEditor.vue -->
<template>
  <el-page-header :title="file?.file_name" @back="$router.back()">
    <template #extra>
      <el-button-group>
        <el-button @click="goVersions">版本历史</el-button>
        <el-button type="primary" :loading="saving" @click="save">
          <kbd>Ctrl+S</kbd> 保存
        </el-button>
      </el-button-group>
    </template>
  </el-page-header>

  <el-form inline class="meta">
    <el-form-item label="语言">
      <el-select v-model="lang" style="width: 180px">
        <el-option v-for="l in LANGUAGES" :key="l" :label="l" :value="l" />
      </el-select>
    </el-form-item>
    <el-form-item label="版本">
      <el-tag>v{{ file?.version_no }}</el-tag>
    </el-form-item>
  </el-form>

  <MonacoEditor v-model="content" :language="lang" height="70vh" />

<script setup lang="ts">
import { ref, onMounted } from 'vue'
import { useRoute, useRouter } from 'vue-router'
import { ElMessage, ElMessageBox } from 'element-plus'
import MonacoEditor from '@/components/editor/MonacoEditor.vue'
import { codeFileApi } from '@/api/codeFile'
import { LANGUAGES } from '@/constants/language'

const route = useRoute()
const router = useRouter()
const id = Number(route.params.id)

const file = ref<any>(null)
const content = ref('')
const lang = ref('python')
const saving = ref(false)

onMounted(async () => {
  file.value = await codeFileApi.detail(id)
  content.value = file.value.content
  lang.value = file.value.language
  window.addEventListener('keydown', onKey)
})

function onKey(e: KeyboardEvent) {
  if ((e.ctrlKey || e.metaKey) && e.key === 's') {
    e.preventDefault()
    save()
  }
}

async function save() {
  const { value: change_desc } = await ElMessageBox.prompt('修改说明(选填)', '保存版本', { inputPlaceholder: '例如:修复登录校验' })
    .catch(() => ({ value: '' }))
  saving.value = true
  try {
    const r = await codeFileApi.update(id, { content: content.value, change_desc })
    file.value.version_no = r.version_no
    ElMessage.success(`已保存,版本 v${r.version_no}`)
  } finally {
    saving.value = false
  }
}

function goVersions() {
  router.push(`/projects/${route.params.projectId}/files/${id}/versions`)
}
</script>

<style scoped>
.meta { margin: 12px 0; }
</style>
```

```vue
import { computed, onMounted, ref } from 'vue'
import { useRouter } from 'vue-router'
import { ElMessage } from 'element-plus'
import { projectApi } from '@/api/project'
import { codeFileApi } from '@/api/codeFile'
import { reviewApi } from '@/api/review'

const router = useRouter()
const projects = ref<any[]>([])
const files = ref<any[]>([])
const projectId = ref<number | null>(null)
const selectedIds = ref<number[]>([])
const reviewType = ref<'quick' | 'standard' | 'full'>('standard')
const loading = ref(false)

onMounted(async () => {
  const r = await projectApi.list({ page: 1, page_size: 100 })
  projects.value = r.items
})

async function loadFiles(pid: number) {
  const r = await codeFileApi.list({ project_id: pid, page: 1, page_size: 100 })
  files.value = r.items
}

function onSel(rows: any[]) { selectedIds.value = rows.map((x) => x.id) }
const canStart = computed(() => projectId.value && selectedIds.value.length > 0)

async function start() {
  if (selectedIds.value.length > 50) return ElMessage.warning('最多 50 个文件')
  loading.value = true
  try {
    const t = await reviewApi.start({
      project_id: projectId.value!,
      file_ids: selectedIds.value,
      review_type: reviewType.value,
    })
```

```vue
<!-- src/components/chart/BaseChart.vue -->

import { onMounted, onBeforeUnmount, ref, watch, shallowRef } from 'vue'
import * as echarts from 'echarts'

const props = defineProps<{ title?: string; option: echarts.EChartsOption }>()
const el = ref<HTMLDivElement>()

  inst.value = echarts.init(el.value!)
  inst.value.setOption(props.option)
  window.addEventListener('resize', resize)

watch(() => props.option, (opt) => inst.value?.setOption(opt, true), { deep: true })

onBeforeUnmount(() => { window.removeEventListener('resize', resize); inst.value?.dispose() })
</script>
```

## 11. 下载文件 Hook

```ts
// src/composables/useDownload.ts
import { download } from '@/api/http'

export async function useDownload(url: string, filename: string, params?: object) {
  const resp = await download(url, params)
  const blob = new Blob([resp.data])
  const a = document.createElement('a')
  a.href = URL.createObjectURL(blob)
  a.download = filename
  a.click()
  URL.revokeObjectURL(a.href)
}
```

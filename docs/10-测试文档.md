```python
import pytest
from app.services import auth_service
from app.schemas.auth import RegisterIn
from app.core.exceptions import ConflictError, AuthError, ForbiddenError


def test_register_creates_user(db):
    user = auth_service.register(db, RegisterIn(username="alice", password="alice123"))
    assert user.id is not None
    assert user.password != "alice123"
    assert user.role == "user"
    assert user.status == 1

    auth_service.register(db, RegisterIn(username="alice", password="alice123"))
    with pytest.raises(ConflictError):
        auth_service.register(db, RegisterIn(username="alice", password="x12345"))


def test_login_success(db):
    auth_service.register(db, RegisterIn(username="alice", password="alice123"))
    token, user = auth_service.login(db, "alice", "alice123")
    assert token
    assert user.username == "alice"


def test_login_wrong_password(db):
    auth_service.register(db, RegisterIn(username="alice", password="alice123"))

        auth_service.login(db, "alice", "wrong")


def test_login_disabled(db, user_factory):
    u = user_factory(username="dave", password="dave1234")
```

```python
import io
from fastapi import UploadFile
from app.services import project_service, code_file_service
from app.schemas.project import ProjectIn


def _make_upload(name: str, content: bytes) -> UploadFile:
    return UploadFile(filename=name, file=io.BytesIO(content))


def test_upload_creates_file_and_version(db, user_factory):
    user = user_factory()
    proj = project_service.create(db, user=user, payload=ProjectIn(project_name="P1", language="python"))
    file_id, lang, ver = code_file_service.upload(
        db=db, user=user, project_id=proj.id,
        upload_file=_make_upload("hello.py", b"print('hi')\n"),
        file_path=None, language=None,
    )
    assert file_id > 0
    assert lang == "python"
    assert ver == 1


def test_update_creates_new_version(db, user_factory):
    user = user_factory()

    fid, _, _ = code_file_service.upload(
        db=db, user=user, project_id=proj.id,
        upload_file=_make_upload("a.py", b"a=1\n"),
        file_path=None, language=None,
    )
    ver = code_file_service.update_content(db=db, user=user, file_id=fid, content="a=2\n", change_desc="改值")
    assert ver == 2
```

```python
# tests/unit/ai/test_code_chunker.py
from app.ai.code_chunker import chunk_code


def test_small_file_not_chunked():
    content = "def a():\n    pass\n"
    chunks = chunk_code(content, "python", threshold=6000)
    assert len(chunks) == 1
    assert chunks[0].start_line == 0


def test_large_python_file_chunks_by_function():
    body = "def foo_{i}():\n    return {i}\n\n"
    content = "".join(body.format(i=i) for i in range(500))
    chunks = chunk_code(content, "python", threshold=2000)
    assert len(chunks) > 1
    # 行号偏移依次递增
    for prev, nxt in zip(chunks, chunks[1:]):
        assert nxt.start_line >= prev.start_line


def test_unknown_language_falls_back_to_lines():
    content = "\n".join([f"line {i}" for i in range(1000)])
    chunks = chunk_code(content, "unknown", threshold=1000)
    assert len(chunks) >= 2
```
